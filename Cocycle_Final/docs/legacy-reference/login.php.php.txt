<?php
session_start();
include 'db_connect.php'; 

if (isset($_POST['login'])) {
    $email    = $_POST['email'];
    $password = $_POST['password'];
    
    // --- [UPDATE] TAMBAHKAN role DAN nama_toko PADA QUERY ---
    $stmt = $conn->prepare("SELECT id, username, email, password, profile_pic, role, nama_toko FROM users WHERE email = ?");

    if ($stmt === false) {
        error_log("DB prepare failed: " . $conn->error);
        echo "<script>alert('Internal server error (DB). Please contact admin.');</script>";
        exit;
    }

    $stmt->bind_param("s", $email);
    $stmt->execute();
    $result = $stmt->get_result();

    if ($result->num_rows === 1) {
        $user = $result->fetch_assoc();

        if (password_verify($password, $user['password'])) {
            
            // Simpan data penting ke Session
            $_SESSION["loggedin"] = true;
            $_SESSION["user_id"] = $user["id"];
            $_SESSION["username"] = $user["username"];
            
            // Simpan ROLE dan NAMA TOKO ke SESSION
            $user_role = isset($user['role']) ? $user['role'] : 'customer';
            $_SESSION["role"] = $user_role;
            $_SESSION["nama_toko"] = isset($user['nama_toko']) ? $user['nama_toko'] : '';
            
            // Cek foto profil
            if (!empty($user["profile_pic"])) {
                $_SESSION["profile_pic"] = $user["profile_pic"];
            } else {
                $_SESSION["profile_pic"] = "assets/default_profile.png"; 
            }
            
            // Cookie Remember Me
            if (isset($_POST['remember'])) {
                setcookie("user_login", $user["username"], time() + (86400 * 30), "/");
            }

            // --- [BARU] LOGIKA REDIRECT BERDASARKAN ROLE ---
            if ($user_role === 'partner') {
                $redirect_target = '/partner/home';
            } else {
                $redirect_target = '/home';
            }

            echo "<script>alert('Login Berhasil sebagai " . ucfirst($user_role) . "!'); window.location.href='$redirect_target';</script>";
            exit;
            
        } else {
            echo "<script>alert('Password salah!');</script>";
        }
    } else {
        echo "<script>alert('Email tidak terdaftar!');</script>";
    }

    $stmt->close();
    $conn->close();
}
